  <script type="text/babel">
    const { useState, useEffect, useRef, useMemo } = React;

    // ── CAMBIA ESTE URL POR EL DE TU DESPLIEGUE ──────────────
    const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbyFQrjcC_LLuDTzdvh_ERtODENeSiwv1N0ATBnq7Qenkb3SK8eHlMWkLcMaZligaluxIw/exec";

    const api = {
      get: async (action) => {
        try { const r = await fetch(`${SCRIPT_URL}?action=${action}`); return await r.json(); }
        catch(e) { return []; }
      },
      post: async (action, data) => {
        try {
          const r = await fetch(SCRIPT_URL, {
            method:"POST", body:JSON.stringify({ action, data }),
            headers:{ "Content-Type":"text/plain;charset=utf-8" }
          });
          return await r.json();
        } catch(e) { return { success:false }; }
      }
    };

    // ============================================================
    //  WIDGET DEL CLIENTE
    // ============================================================
    function ClientWebsite({ allMessages, cannedResponses }) {
      const [isOpen,     setIsOpen]     = useState(false);
      const [newMessage, setNewMessage] = useState("");
      const [showEmojis, setShowEmojis] = useState(false);
      const emojis = ["😀","😂","😊","🥰","😎","🤔","👍","👎","🔥","✨","🎉","🚀","✅","❌","👀","⚡","📡","🔧","🏢","📝"];

      // SISTEMA DE ARRASTRE
      const [pos,        setPos]        = useState(null);
      const [isDragging, setIsDragging] = useState(false);
      const widgetRef = useRef(null);
      const dragRef   = useRef({ startX:0, startY:0, initialRight:0, initialBottom:0, isDragMove:false });

      const handleDragStart = (e) => {
        if (e.type.includes("mouse") && e.button !== 0) return;
        if (e.target.closest("button") && !e.target.closest(".drag-handle")) return;
        const clientX = e.type.includes("mouse") ? e.clientX : e.touches[0].clientX;
        const clientY = e.type.includes("mouse") ? e.clientY : e.touches[0].clientY;
        const rect = widgetRef.current.getBoundingClientRect();
        const initialRight  = window.innerWidth  - rect.right;
        const initialBottom = window.innerHeight - rect.bottom;
        dragRef.current = { startX:clientX, startY:clientY, initialRight, initialBottom, isDragMove:false };
        if (!pos) setPos({ right:initialRight, bottom:initialBottom });
        setIsDragging(true);
      };

      useEffect(() => {
        const handleDragMove = (e) => {
          if (!isDragging) return;
          e.preventDefault();
          const clientX = e.type.includes("mouse") ? e.clientX : e.touches[0].clientX;
          const clientY = e.type.includes("mouse") ? e.clientY : e.touches[0].clientY;
          const dx = clientX - dragRef.current.startX;
          const dy = clientY - dragRef.current.startY;
          if (Math.abs(dx)>3||Math.abs(dy)>3) dragRef.current.isDragMove = true;
          const rect = widgetRef.current.getBoundingClientRect();
          const maxRight  = window.innerWidth  - rect.width;
          const maxBottom = window.innerHeight - rect.height;
          setPos({
            right:  Math.max(0, Math.min(dragRef.current.initialRight  - dx, maxRight)),
            bottom: Math.max(0, Math.min(dragRef.current.initialBottom - dy, maxBottom))
          });
        };
        const handleDragEnd = () => setIsDragging(false);
        if (isDragging) {
          window.addEventListener("mousemove",  handleDragMove, { passive:false });
          window.addEventListener("mouseup",    handleDragEnd);
          window.addEventListener("touchmove",  handleDragMove, { passive:false });
          window.addEventListener("touchend",   handleDragEnd);
        }
        return () => {
          window.removeEventListener("mousemove",  handleDragMove);
          window.removeEventListener("mouseup",    handleDragEnd);
          window.removeEventListener("touchmove",  handleDragMove);
          window.removeEventListener("touchend",   handleDragEnd);
        };
      }, [isDragging]);

      const toggleOpen = (e) => {
        if (dragRef.current?.isDragMove) { e.preventDefault(); e.stopPropagation(); return; }
        setIsOpen(prev => !prev);
      };

      // Efecto ROBUSTO para reajustar la posición (incluye móviles y teclado virtual)
      useEffect(() => {
        if (isOpen) {
          const adjustPos = () => {
            // Usamos visualViewport si está disponible (esencial para detectar el teclado en móviles)
            const vHeight = window.visualViewport ? window.visualViewport.height : window.innerHeight;
            const vWidth  = window.visualViewport ? window.visualViewport.width : window.innerWidth;
            const margin  = 16;
            const isMobile = vWidth < 640;

            // Dimensiones basadas en las clases CSS actuales
            const chatWidth  = isMobile ? vWidth - 48 : 380; 
            const chatHeight = Math.min(550, vHeight * 0.85);

            setPos(currentPos => {
              // Si no hay posición registrada aún, asume la posición inicial del botón
              let r = currentPos ? currentPos.right : 24;
              let b = currentPos ? currentPos.bottom : 112;

              const maxRight  = Math.max(margin, vWidth - chatWidth - margin);
              const maxBottom = Math.max(margin, vHeight - chatHeight - margin);

              if (r > maxRight)  r = maxRight;
              if (b > maxBottom) b = maxBottom;
              if (r < margin) r = margin;
              if (b < margin) b = margin;

              // En móviles, forzamos el centrado horizontal para evitar recortes
              if (isMobile) {
                r = (vWidth - chatWidth) / 2;
              }

              // Actualiza solo si es estrictamente necesario
              if (!currentPos || r !== currentPos.right || b !== currentPos.bottom) {
                return { right: r, bottom: b };
              }
              return currentPos;
            });
          };

          // Pequeño retraso para dar tiempo a que las clases CSS de la ventana abierta se apliquen
          setTimeout(adjustPos, 50);

          // Escuchamos los cambios de tamaño, especialmente útiles para teclados móviles
          window.addEventListener('resize', adjustPos);
          if (window.visualViewport) {
            window.visualViewport.addEventListener('resize', adjustPos);
          }

          return () => {
            window.removeEventListener('resize', adjustPos);
            if (window.visualViewport) {
              window.visualViewport.removeEventListener('resize', adjustPos);
            }
          };
        }
      }, [isOpen]);

      // Detectar usuario logueado en el sitio principal
      const getEliteUser = () => {
        try {
          const params = new URLSearchParams(window.location.search);
          const pName = params.get("nombre") || params.get("name");
          const pId   = params.get("id")     || params.get("email");
          if (pName) return { name:pName, id:pId };
          const userStr = localStorage.getItem("freework_user");
          if (userStr) { const u = JSON.parse(userStr); return { name:u.name||u.nombre, id:u.id||u.email }; }
          const fallback = localStorage.getItem("freework_user_name");
          if (fallback) return { name:fallback, id:localStorage.getItem("freework_user_id") };
        } catch(e) {}
        return null;
      };

      const [clientName,   setClientName]   = useState(() => getEliteUser()?.name || "");
      const [isRegistered, setIsRegistered] = useState(() => !!getEliteUser()?.name);
      const [hasUnread,    setHasUnread]    = useState(false);
      const messagesEndRef = useRef(null);
      const prevMsgCount   = useRef(0);

      // Polling para detectar usuario logueado dinámicamente
      useEffect(() => {
        const interval = setInterval(() => {
          if (!isRegistered) {
            const user = getEliteUser();
            if (user?.name) { setClientName(user.name); setIsRegistered(true); }
          }
        }, 1000);
        return () => clearInterval(interval);
      }, [isRegistered]);

      // ID único del cliente — persiste en localStorage
      const [clientId] = useState(() => {
        const user = getEliteUser();
        let id = user?.id || localStorage.getItem("chat_client_id");
        if (!id) {
          id = "cliente_" + Date.now() + "_" + Math.floor(Math.random() * 100000);
          localStorage.setItem("chat_client_id", id);
        }
        return id;
      });

      const myMessages = useMemo(() =>
        allMessages
          .filter(m => m.conversationId === clientId)
          .sort((a,b) => Number(a.timestamp) - Number(b.timestamp)),
        [allMessages, clientId]
      );

      useEffect(() => {
        if (myMessages.length > prevMsgCount.current) {
          const last = myMessages[myMessages.length - 1];
          if (last.sender !== "client" && !isOpen) setHasUnread(true);
        }
        prevMsgCount.current = myMessages.length;
      }, [myMessages, isOpen]);

      useEffect(() => { if (isOpen) setHasUnread(false); }, [isOpen]);
      useEffect(() => { messagesEndRef.current?.scrollIntoView({ behavior:"smooth" }); }, [myMessages, isOpen]);

      // Registro + mensaje ONBOARDING
      const handleRegister = () => {
        if (!clientName.trim()) return;
        setIsRegistered(true);

        // Bot client-side: mensaje de bienvenida (título = ONBOARDING)
        const onboard = cannedResponses?.find(r => r.title?.toUpperCase().trim() === "ONBOARDING");
        if (onboard) {
          setTimeout(() => {
            api.post("sendMessage", {
              chatId:     clientId,
              senderRole: "admin",
              senderName: "SISTEMA",
              text:       onboard.text.replace(/\{nombre\}/gi, clientName)
            });
          }, 1200);
        }
      };

      // Envío de mensaje + bot client-side por keywords en títulos de Chat_Respuestas
      const handleSendMessage = async (e) => {
        e.preventDefault();
        if (!newMessage.trim() || !clientName) return;
        const text = newMessage.trim();
        setNewMessage(""); setShowEmojis(false);

        // 1. Enviar mensaje del cliente
        await api.post("sendMessage", {
          chatId:     clientId,
          senderRole: "client",
          senderName: clientName,
          text
        });

        // 2. Bot client-side — keywords en títulos de Chat_Respuestas
        const textLower = text.toLowerCase();
        const matched = cannedResponses?.find(r => {
          if (!r.title || r.title.toUpperCase().trim() === "ONBOARDING") return false;
          const keywords = r.title.split(",").map(k => k.trim().toLowerCase()).filter(Boolean);
          return keywords.some(k => k.length > 2 && textLower.includes(k));
        });
        if (matched) {
          setTimeout(() => {
            api.post("sendMessage", {
              chatId:     clientId,
              senderRole: "admin",
              senderName: "SISTEMA",
              text:       matched.text.replace(/\{nombre\}/gi, clientName)
            });
          }, 1500);
        }
      };

      const addEmoji = (emoji) => setNewMessage(prev => prev + emoji);

      return (
        <div className={`h-screen w-full relative overflow-hidden flex flex-col font-sans pointer-events-none ${isDragging ? "dragging-active" : ""}`}>

          {/* CONTENEDOR ARRASTRABLE */}
          <div
            ref={widgetRef}
            className={`absolute z-50 flex flex-col items-end pointer-events-auto ${!pos ? "bottom-28 right-6" : ""}`}
            style={pos ? { right:pos.right, bottom:pos.bottom, transition:isDragging ? "none" : "bottom 0.2s, right 0.2s" } : {}}
          >

            {/* BOTÓN FLOTANTE */}
            {!isOpen && (
              <button
                onMouseDown={handleDragStart}
                onTouchStart={handleDragStart}
                onClick={toggleOpen}
                className={`drag-handle btn-premium group flex items-center bg-gradient-to-r from-[#b233ff]/80 to-[#ff4df8]/80 backdrop-blur-md text-white rounded-full transition-all duration-300 outline-none relative overflow-hidden border border-white/30 cursor-move ${isDragging ? "scale-95" : "hover:pr-8 hover:scale-105"}`}
                style={{ WebkitTapHighlightColor:"transparent" }}
              >
                {hasUnread && (
                  <span className="absolute top-2 right-2 bg-rose-500 text-white w-5 h-5 flex items-center justify-center rounded-full animate-bounce shadow-lg border-2 border-white z-50">
                    <i className="fa-solid fa-bell text-[9px]"></i>
                  </span>
                )}
                <div className="w-12 h-12 flex items-center justify-center shrink-0 relative z-10 pointer-events-none">
                  <i className="fa-solid fa-location-dot text-[20px] group-hover:scale-110 transition-transform duration-500 drop-shadow-md"></i>
                </div>
                <div className="max-w-0 overflow-hidden opacity-0 group-hover:max-w-[220px] group-hover:opacity-100 transition-all duration-500 ease-in-out flex items-center whitespace-nowrap relative z-10 pointer-events-none">
                  <span className="font-black text-[11px] sm:text-xs uppercase tracking-[0.2em] pr-2 drop-shadow-md">
                    Validar Factibilidad
                  </span>
                </div>
              </button>
            )}

            {/* VENTANA DEL CHAT */}
            {isOpen && (
              <div className="w-[calc(100vw-3rem)] sm:w-[380px] h-[550px] max-h-[85vh] bg-white/85 backdrop-blur-2xl rounded-[2rem] shadow-[0_25px_50px_-12px_rgba(178,51,255,0.25)] flex flex-col overflow-hidden z-50 border border-white/60 animate-slide-up">

                {/* CABECERA */}
                <div
                  className="drag-handle bg-gradient-to-r from-slate-900 to-slate-800 p-5 flex justify-between items-center relative shrink-0 shadow-md z-10 cursor-move select-none"
                  onMouseDown={handleDragStart}
                  onTouchStart={handleDragStart}
                >
                  <div className="flex items-center gap-3 sm:gap-4 pointer-events-none">
                    <div className="w-10 h-10 bg-gradient-to-br from-[#b233ff] to-[#ff4df8] rounded-2xl flex items-center justify-center text-white shadow-lg shadow-purple-500/30">
                      <i className="fa-solid fa-location-dot text-lg"></i>
                    </div>
                    <div>
                      <h3 className="font-black text-white text-base uppercase tracking-tight italic drop-shadow-sm">Terminal BOT</h3>
                      <p className="text-slate-300 text-[9px] font-bold uppercase tracking-[0.15em] mt-0.5 opacity-80">Sistema de Factibilidad</p>
                    </div>
                  </div>
                  <button onClick={toggleOpen} className="text-white/50 hover:text-white hover:bg-white/10 hover:rotate-90 w-8 h-8 rounded-full flex items-center justify-center transition-all duration-300">
                    <i className="fa-solid fa-xmark text-xl pointer-events-none"></i>
                  </button>
                </div>

                {/* PANTALLA DE REGISTRO */}
                {!isRegistered ? (
                  <div className="flex-1 p-8 sm:p-10 bg-transparent flex flex-col justify-center items-center text-center relative">
                    <div className="absolute inset-0 bg-gradient-to-b from-transparent to-purple-50/30 -z-10"></div>
                    <div className="w-20 h-20 sm:w-24 sm:h-24 bg-white/80 border border-white shadow-[0_15px_35px_-5px_rgba(178,51,255,0.15)] text-[#b233ff] rounded-[2.5rem] flex items-center justify-center mb-6 sm:mb-8 shrink-0">
                      <i className="fa-solid fa-fingerprint text-4xl sm:text-5xl opacity-80"></i>
                    </div>
                    <h4 className="text-xl sm:text-2xl font-black text-slate-800 mb-2 uppercase tracking-tight">Acceso Requerido</h4>
                    <p className="text-xs text-slate-500 mb-6 sm:mb-8 font-medium leading-relaxed px-4">Identifícate para iniciar el escaneo de factibilidad técnica.</p>
                    <input
                      type="text"
                      placeholder="Tu nombre completo"
                      value={clientName}
                      onChange={e => setClientName(e.target.value)}
                      onKeyDown={e => { if (e.key==="Enter") handleRegister(); }}
                      className="w-full bg-white/80 border border-white/60 shadow-inner rounded-2xl px-5 sm:px-6 py-3.5 sm:py-4 text-sm mb-4 outline-none focus:ring-2 focus:ring-[#b233ff]/40 transition-all font-semibold"
                    />
                    <button
                      onClick={handleRegister}
                      className="w-full bg-gradient-to-r from-[#b233ff] to-[#ff4df8] text-white py-3.5 sm:py-4 rounded-2xl font-black text-[10px] uppercase tracking-[0.2em] shadow-[0_10px_20px_-5px_rgba(178,51,255,0.4)] hover:shadow-[0_15px_25px_-5px_rgba(178,51,255,0.6)] active:scale-95 transition-all duration-300">
                      Inicializar Escaneo
                    </button>
                  </div>
                ) : (
                  <>
                    {/* MENSAJES */}
                    <div className="flex-1 overflow-y-auto p-4 sm:p-6 bg-slate-50/40 flex flex-col gap-5 relative">
                      <div className="absolute inset-0 bg-gradient-to-b from-purple-50/20 to-transparent pointer-events-none"></div>

                      {/* Mensaje inicial estático de bienvenida */}
                      <div className="flex flex-col items-start relative z-10">
                        <div className="px-4 sm:px-5 py-3.5 rounded-3xl rounded-tl-sm bg-white/90 backdrop-blur-md border border-white shadow-sm text-slate-700 text-sm">
                          <p className="font-black text-[10px] text-transparent bg-clip-text bg-gradient-to-r from-[#b233ff] to-[#ff4df8] uppercase tracking-widest border-b border-slate-100 pb-2 mb-3">CONEXIÓN ESTABLECIDA</p>
                          <p className="font-bold text-slate-900 mb-1">HOLA, {clientName.toUpperCase()}</p>
                          <p className="text-slate-500 leading-relaxed">Envía la dirección para validar la cobertura de fibra óptica e infraestructura.</p>
                        </div>
                      </div>

                      {myMessages.map(msg => {
                        const isMe     = msg.sender === "client";
                        const isSystem = msg.senderName === "SISTEMA" || msg.senderName === "Bot Automático";

                        // Mensajes del bot — burbuja amigable con ícono
                        if (isSystem) return (
                          <div key={msg.id} className="flex items-start gap-2 self-start max-w-[85%] relative z-10">
                            <div className="w-7 h-7 rounded-full bg-emerald-100 flex items-center justify-center shrink-0 mt-1 border border-emerald-200">
                              <i className="fa-solid fa-robot text-emerald-500 text-xs"></i>
                            </div>
                            <div className="flex flex-col items-start">
                              <div className="bg-white/90 backdrop-blur-md border border-white text-slate-700 px-4 sm:px-5 py-3 sm:py-3.5 rounded-3xl rounded-tl-sm text-[13px] leading-relaxed shadow-sm whitespace-pre-wrap">
                                {msg.text}
                              </div>
                            </div>
                          </div>
                        );

                        // Mensajes normales (cliente y agente)
                        return (
                          <div key={msg.id} className={`flex flex-col max-w-[85%] relative z-10 ${isMe ? "self-end items-end" : "self-start items-start"}`}>
                            <div className={`px-4 sm:px-5 py-3 sm:py-3.5 rounded-3xl text-[13px] font-medium leading-relaxed shadow-sm whitespace-pre-wrap ${
                              isMe
                                ? "bg-gradient-to-r from-[#b233ff] to-[#ff4df8] text-white rounded-br-sm shadow-[0_5px_15px_-5px_rgba(178,51,255,0.4)]"
                                : "bg-white/90 backdrop-blur-md border border-white text-slate-700 rounded-bl-sm"
                            }`}>
                              {msg.text}
                            </div>
                          </div>
                        );
                      })}
                      <div ref={messagesEndRef} className="h-2"/>
                    </div>

                    {/* INPUT */}
                    <form onSubmit={handleSendMessage} className="p-3 sm:p-4 bg-white/70 backdrop-blur-xl border-t border-white/60 flex gap-3 shrink-0 relative z-20 items-center">
                      <div className="relative flex-1 flex items-center bg-white/80 border border-white shadow-inner rounded-full focus-within:bg-white focus-within:ring-2 focus-within:ring-[#b233ff]/30 transition-all">
                        <button type="button" onClick={()=>setShowEmojis(!showEmojis)}
                          className={`pl-4 pr-2 transition-colors duration-300 ${showEmojis ? "text-[#b233ff]" : "text-slate-400 hover:text-[#b233ff]"}`}>
                          <i className="fa-regular fa-face-smile text-lg"></i>
                        </button>
                        <input
                          type="text"
                          value={newMessage}
                          onChange={e => setNewMessage(e.target.value)}
                          placeholder="Escribe la dirección..."
                          className="flex-1 bg-transparent py-3.5 sm:py-4 pr-5 text-sm outline-none font-medium text-slate-700 w-full"
                        />
                        {showEmojis && (
                          <div className="absolute bottom-[calc(100%+12px)] left-0 w-[240px] bg-white/95 backdrop-blur-3xl border border-white/60 shadow-[0_15px_35px_-5px_rgba(178,51,255,0.3)] rounded-3xl p-3 grid grid-cols-5 gap-1.5 z-50 animate-slide-up">
                            {emojis.map(emoji => (
                              <button key={emoji} type="button" onClick={()=>addEmoji(emoji)}
                                className="text-xl hover:bg-[#b233ff]/10 rounded-xl p-1.5 transition-all hover:scale-125 active:scale-95 flex items-center justify-center">
                                {emoji}
                              </button>
                            ))}
                          </div>
                        )}
                      </div>
                      <button type="submit" disabled={!newMessage.trim()}
                        className="bg-gradient-to-r from-[#b233ff] to-[#ff4df8] text-white w-11 h-11 sm:w-12 sm:h-12 flex items-center justify-center rounded-full hover:shadow-[0_8px_20px_-5px_rgba(178,51,255,0.5)] disabled:opacity-40 disabled:grayscale flex-shrink-0 shadow-md active:scale-95 transition-all duration-300">
                        <i className="fa-solid fa-paper-plane text-sm translate-x-[-1px] translate-y-[1px]"></i>
                      </button>
                    </form>
                  </>
                )}
              </div>
            )}
          </div>
        </div>
      );
    }

    function App() {
      const [allMessages,     setAllMessages]     = useState([]);
      const [cannedResponses, setCannedResponses] = useState([]);

      useEffect(() => {
        let isMounted = true;
        const loadData = async () => {
          const [msgsRaw, cannedRaw] = await Promise.all([
            api.get("getMessages"),
            api.get("getCannedResponses")
          ]);
          if (!isMounted) return;
          if (Array.isArray(msgsRaw)) setAllMessages(msgsRaw.map(m => ({
            id:             m["MENSAJE_ID"],
            conversationId: m["CHAT_ID"],
            sender:         m["ROL_EMISOR"],
            senderName:     m["NOMBRE_EMISOR"],
            text:           m["TEXTO"],
            timestamp:      m["TIMESTAMP"]
          })));
          if (Array.isArray(cannedRaw)) setCannedResponses(cannedRaw.map(r => ({
            id:    r["ID_RESPUESTA"],
            title: r["TITULO"],
            text:  r["TEXTO"]
          })));
        };
        loadData();
        const interval = setInterval(loadData, 4000);
        return () => { isMounted = false; clearInterval(interval); };
      }, []);

      return <ClientWebsite allMessages={allMessages} cannedResponses={cannedResponses}/>;
    }

    ReactDOM.createRoot(document.getElementById("root")).render(<App/>);
  </script>
